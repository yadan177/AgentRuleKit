## 1. 🔴 硬约束 · 这些绝对不要写

### 1.1 var 变量提升

```javascript
// ❌ 错误：var 变量提升,作用域混乱,值可能变成 undefined
console.log(a); // 输出 undefined(不是报错)
var a = 1;

// 循环里的 var 全部共享同一个作用域
for (var i = 0; i < 3; i++) {
  setTimeout(() => console.log(i), 0); // 输出 3, 3, 3 不是 0, 1, 2
}

// ✅ 正确：统一用 const,需要改的用 let
console.log(b); // 直接报错 ReferenceError(期望行为)
const b = 1;

for (let j = 0; j < 3; j++) {
  setTimeout(() => console.log(j), 0); // 输出 0, 1, 2
}
```

> 📌 规则：**永远不要用 var，全部用 const / let**

### 1.2 == 隐式类型转换的 10 大坑

```javascript
// ❌ 错误：== 有疯狂的隐式类型转换规则,根本记不住
'' == '0'           // false
0 == ''             // true  💥
0 == '0'            // true  💥

false == 'false'    // false
false == '0'        // true  💥

false == undefined  // false
false == null       // false
null == undefined   // true  💥

' \t\r\n ' == 0    // true  💥 空白字符串等于0？

1 == true           // true
2 == true           // false 💥 那1和2到底谁是true？

// ✅ 正确：永远只用 === / !==
'' === '0'          // false
0 === ''            // false
0 === '0'           // false
false === '0'       // false
null === undefined  // false
```

> 📌 规则：**永远不要用 == / !=，全部用 === / !==**

### 1.3 forEach 滥用（该用专门的数组方法）

```javascript
// ❌ 错误：forEach 写了一堆手动判断,可读性差、容易漏
const numbers = [1, 2, 3, 4, 5];
let sum = 0;
let hasEven = false;
const evens = [];

numbers.forEach(n => {
  sum += n;                        // 求和应该用 reduce
  if (n % 2 === 0) hasEven = true; // 判断有没有应该用 some
  if (n % 2 === 0) evens.push(n);  // 过滤应该用 filter
});

// ✅ 正确：用专门的数组方法,意图明确
const sum2 = numbers.reduce((acc, n) => acc + n, 0);
const hasEven2 = numbers.some(n => n % 2 === 0);
const evens2 = numbers.filter(n => n % 2 === 0);
const allEven = numbers.every(n => n % 2 === 0);
const firstEven = numbers.find(n => n % 2 === 0);
```

> 📌 规则：
> - 求和 / 聚合 → `reduce`
> - 判断有没有 → `some`
> - 判断全部 → `every`
> - 过滤 → `filter`
> - 找第一个 → `find`
> - 转换每个元素 → `map`
> - **只有真正需要 side effect 时才用 forEach**

### 1.4 async/await 在 for 循环里串行请求

```javascript
// ❌ 错误：本来可以并行的请求,被写成串行,慢3倍
const userIds = [1, 2, 3];
const users = [];

for (const id of userIds) {
  const user = await fetchUser(id); // 每个请求等上一个完成才发
  users.push(user);
}
// 总耗时：t1 + t2 + t3

// ✅ 正确：用 Promise.all 并行请求
const users2 = await Promise.all(userIds.map(id => fetchUser(id)));
// 总耗时：max(t1, t2, t3)
```

> 📌 规则：**独立请求默认并行，只有真的有依赖关系时才用串行**

### 1.5 for 循环里定义函数（闭包坑）

```javascript
// ❌ 错误：每个循环共享同一个变量 i,所有函数拿到的都是最后那个值
const buttons = document.querySelectorAll('button');
for (var i = 0; i < buttons.length; i++) {
  buttons[i].addEventListener('click', function() {
    console.log('You clicked button #' + i); // 永远输出 buttons.length
  });
}

// ✅ 正确：用 let 块级作用域
for (let j = 0; j < buttons.length; j++) {
  buttons[j].addEventListener('click', function() {
    console.log('You clicked button #' + j); // 正确输出 0, 1, 2...
  });
}
```

> 📌 规则：循环计数器永远用 let，不要用 var

### 1.6 闭包内存泄漏（事件监听不清理）

```javascript
// ❌ 错误：组件销毁了,但事件监听还在,整个闭包都泄漏了
function Component() {
  const data = { /* 很大的对象 */ };

  window.addEventListener('resize', () => {
    console.log(data); // data 被闭包引用,永远无法 GC
  });
  // 组件销毁后,resize 监听还在,data 永远泄漏
}

// ✅ 正确：清理函数/组件销毁时 removeEventListener
function Component2() {
  const data = { /* 很大的对象 */ };

  const handleResize = () => console.log(data);
  window.addEventListener('resize', handleResize);

  // 返回清理函数,组件销毁时调用
  return () => window.removeEventListener('resize', handleResize);
}
```

> 📌 规则：**addEventListener 必须有对应的 removeEventListener**

### 1.7 this 绑定 5 种常见坑

```javascript
const obj = {
  value: 10,
  getValue: function() { return this.value; },
  getValueArrow: () => this.value, // 箭头函数没有自己的 this
};

// ❌ 坑1：方法拿出来单独调用,this 变成 undefined(严格模式)
const getValue = obj.getValue;
getValue(); // undefined(不是 10！)

// ❌ 坑2：箭头函数作为对象方法,this 指向外层不是对象
obj.getValueArrow(); // undefined(浏览器里是 window)

// ❌ 坑3：setTimeout 里调用对象方法,this 丢失
setTimeout(obj.getValue, 1000); // 回调里 this 不是 obj

// ❌ 坑4：回调函数里 this 变了
const button = document.querySelector('button');
button.addEventListener('click', obj.getValue); // 回调里 this 是 button,不是 obj

// ❌ 坑5：class 方法作为 props 传给 React 组件,this 丢失
<Button onClick={obj.getValue} /> // this 丢失

// ✅ 正确：bind / 箭头函数 / 类属性箭头函数
const getValue2 = obj.getValue.bind(obj); // bind 绑定
getValue2(); // 10

// 或者定义时就用箭头函数
const obj2 = {
  value: 10,
  getValue: () => obj2.value, // 直接闭包引用对象本身
};
```

> 📌 规则：**需要传出去当回调的方法，必须 bind 或者用箭头函数**

### 1.8 try-catch 吞错误

```javascript
// ❌ 错误：catch 什么都不做,出了问题根本不知道
try {
  JSON.parse(someData);
} catch (e) {
  // 空 catch = 吞错误 = 噩梦级调试
}

// ❌ 更糟：只 log 不 throw,上层不知道出错了
try {
  JSON.parse(someData);
} catch (e) {
  console.log(e); // 只打个 log,上层根本感知不到出错了
}

// ✅ 正确：要么不 catch,要么正确处理,处理不了就 rethrow
try {
  JSON.parse(someData);
} catch (e) {
  console.error('JSON parse failed:', e); // 打完整 error stack
  // 要么 fallback
  return defaultValue;
  // 要么 rethrow 让上层处理
  // throw e;
}
```

> 📌 规则：**绝对不要写空 catch，不要只 log 不处理**

### 1.9 prototype 乱加方法（污染原型）

```javascript
// ❌ 错误：在内置对象 prototype 上加方法,污染全局
Array.prototype.sum = function() {
  return this.reduce((a, b) => a + b, 0);
};
// 后果：所有数组都多了个 sum 方法,for...in 会遍历出来,库之间冲突

// ✅ 正确：写工具函数,不要改原型
function sum(arr) {
  return arr.reduce((a, b) => a + b, 0);
}
```

> 📌 规则：**永远不要修改内置对象的 prototype**

### 1.10 魔数 / 魔字符串

```javascript
// ❌ 错误：代码里出现不知道什么意思的数字/字符串
if (user.role === 3) { /* 3 是什么？没人知道 */ }
if (status === 'pending_approval_step2') { /* 拼错一个字母就完了 */ }

// ✅ 正确：用常量 / 枚举
const ROLES = {
  ADMIN: 1,
  EDITOR: 2,
  VIEWER: 3,
};
const STATUS = {
  PENDING: 'pending',
  APPROVED: 'approved',
  REJECTED: 'rejected',
};

if (user.role === ROLES.VIEWER) { /* 一目了然 */ }
if (status === STATUS.PENDING) { /* 拼错直接报 undefined,好排查 */ }
```

> 📌 规则：**不要在业务逻辑里直接写数字/字符串，统一定义常量**

## 2. 🟡 推荐 · 这些写法尽量避免

### 2.11 深层嵌套回调（回调地狱）

```javascript
// ❌ 错误：三层以上嵌套,根本读不懂
fetchUser(1, (err, user) => {
  if (err) throw err;
  fetchPosts(user.id, (err, posts) => {
    if (err) throw err;
    fetchComments(posts[0].id, (err, comments) => {
      if (err) throw err;
      // 三层了,继续嵌套...
    });
  });
});

// ✅ 正确：async/await + Promise 扁平化
const user = await fetchUser(1);
const posts = await fetchPosts(user.id);
const comments = await fetchComments(posts[0].id);
```

### 2.12 函数参数超过 3 个

```javascript
// ❌ 错误：参数多了记不住顺序,容易传错
function createUser(name, email, password, role, status, avatar) {
  // 调用时根本记不住顺序
}
createUser('Tom', 'tom@example.com', '123456', 3, 'active', 'xxx.jpg'); // 错了根本看不出来

// ✅ 正确：用对象传参,顺序不重要,还能有默认值
function createUser({ name, email, password, role = ROLES.VIEWER, status, avatar }) {
  // 一目了然
}
createUser({
  name: 'Tom',
  email: 'tom@example.com',
  password: '123456',
  status: 'active',
  avatar: 'xxx.jpg',
});
```

### 2.13 巨型函数（超过 50 行）

```javascript
// ❌ 错误：一个函数干了 N 件事,无法测试、无法复用
function handleSubmit(formData) {
  // 10 行验证
  // 20 行格式化
  // 15 行 API 调用
  // 10 行错误处理
  // 15 行跳转 / toast / 埋点
  // 总共 70 行 = 没人敢改
}

// ✅ 正确：拆分,每个函数只做一件事
function validateForm(data) { /* 只做验证 */ }
function formatFormData(data) { /* 只做格式化 */ }
function submitApi(data) { /* 只调 API */ }
function handleError(err) { /* 只处理错误 */ }

function handleSubmit(formData) {
  const valid = validateForm(formData);
  const formatted = formatFormData(valid);
  try {
    await submitApi(formatted);
  } catch (e) {
    handleError(e);
  }
}
```

## 3. 📌 总结：JS 写代码三原则

1. **不要自作聪明**：不要写你觉得"巧妙"的隐式转换/原型链 trick，越简单越好
2. **意图明确**：用 `reduce` 就是求和，用 `some` 就是判断有没有，代码本身要说话
3. **不要吞错误**：catch 了要么处理，要么抛出去，别装没看见
